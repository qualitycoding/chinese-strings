cmake_minimum_required(VERSION 3.25)
project(chinese_strings VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CS_BUILD_TESTS "Build tests" ON)
option(CS_BUILD_PLUGIN "Build JUCE-dependent targets" ON)
option(CS_FUZZ "Build libFuzzer targets (clang only)" OFF)
include(FetchContent)
# Pins: plan/deps.lock.json (D-001, D-005, D-010)
FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG 317ac1ed4c0bb6e6b91eafc817e05c488feffcb3)
FetchContent_Declare(tuning_library GIT_REPOSITORY https://github.com/surge-synthesizer/tuning-library.git GIT_TAG 48422e2f014fcda8dd5d1a4678bb2674faf3bb3e)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG be29c81492b6151c8ea8d14c840e1311963b3a83)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB CS_INSTRUMENT_JSON CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/data/instruments/*.json ${CMAKE_SOURCE_DIR}/data/bodies/*.json)
set(CS_GEN_TABLES ${CMAKE_BINARY_DIR}/generated/InstrumentTables.cpp)
add_custom_command(OUTPUT ${CS_GEN_TABLES}
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/gen_instruments.py ${CMAKE_SOURCE_DIR}/data ${CS_GEN_TABLES}
  DEPENDS ${CMAKE_SOURCE_DIR}/tools/gen_instruments.py ${CS_INSTRUMENT_JSON} VERBATIM)
add_library(cs_core STATIC core/src/Stubs.cpp core/src/StubsS005.cpp core/src/Instruments.cpp core/src/FractionalDelay.cpp core/src/BowJunction.cpp core/src/ModalBank.cpp ${CS_GEN_TABLES})
target_include_directories(cs_core PUBLIC core/include PRIVATE core/src)

if(CS_BUILD_PLUGIN)
  FetchContent_MakeAvailable(JUCE)
  add_library(cs_plugin_core STATIC plugin/core/src/StateCodecStub.cpp)
  target_include_directories(cs_plugin_core PUBLIC plugin/core/include)
  target_compile_definitions(cs_plugin_core PUBLIC JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_STANDALONE_APPLICATION=1)
  target_link_libraries(cs_plugin_core PUBLIC cs_core PRIVATE juce::juce_core juce::juce_data_structures)
endif()

if(CS_BUILD_TESTS)
  FetchContent_MakeAvailable(Catch2)
  enable_testing()
  add_subdirectory(tests)
endif()
